### Fix 2: Add Caching to Everything

```python
import streamlit as st

# Cache data loading (persists across reruns)
@st.cache_data(ttl=3600)  # Cache for 1 hour
def load_watershed_boundaries():
    return gpd.read_file("watersheds.geojson")

# Cache expensive computations
@st.cache_data
def calculate_runoff(area, cn, rainfall):
    # SCS-CN calculation
    S = (1000 / cn) - 10
    if rainfall > 0.2 * S:
        Q = ((rainfall - 0.2 * S) ** 2) / (rainfall + 0.8 * S)
    else:
        Q = 0
    return Q * area

# Cache resources (models, connections)
@st.cache_resource
def load_model():
    return pickle.load(open("watershed_model.pkl", "rb"))
```

## Quick Diagnostic Script

Add this at the TOP of your app.py to identify the bottleneck:

```python
import streamlit as st
import time

st.set_page_config(page_title="Watershed Simulator", page_icon="🌊")

# Checkpoint timing
checkpoints = []

def checkpoint(name):
    checkpoints.append((name, time.time()))
    if len(checkpoints) > 1:
        elapsed = checkpoints[-1][1] - checkpoints[-2][1]
        st.sidebar.write(f"✓ {name}: {elapsed:.2f}s")
    else:
        st.sidebar.write(f"▶ Starting: {name}")

checkpoint("Page config loaded")

# Now import your heavy modules
```
